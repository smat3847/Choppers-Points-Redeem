function handleRequest(data) {
  var ss = SpreadsheetApp.openById('1EWpvowiEHcI8vqVwd4hnYpfNyHRLkWVCcBy8ZtGXRdM');
  var sheet = ss.getSheetByName('Customer Rewards Data') || ss.getSheets()[0];
  var rowNum = Number(data.sl) + 1; // Row 1 is Header, Row 2 is SL 1
  if (Number(sheet.getRange(rowNum, 1).getValue()) !== Number(data.sl)) {
    rowNum = Number(data.sl) + 4;
  }
  sheet.getRange(rowNum, 5).setValue(Number(data.spentPoints));
  sheet.getRange(rowNum, 6).setValue(Number(data.balancePoints));
  sheet.getRange(rowNum, 7).setValue(Number(data.balanceAmt));
  var logSheet = ss.getSheetByName('Saved Redemptions') || ss.insertSheet('Saved Redemptions');
  if (logSheet.getLastRow() === 0) {
    logSheet.appendRow(['Timestamp', 'SL.', 'Customer ID', 'Customer Name', 'Previous Balance Amt', 'Redeemed Amt', 'Updated Balance Amt']);
  }
  logSheet.appendRow([data.time || new Date().toLocaleTimeString(), Number(data.sl), data.customerId, data.customerName, Number(data.prevBalance), Number(data.redeemedAmt), Number(data.newBalance)]);

}
function doPost(e) {
